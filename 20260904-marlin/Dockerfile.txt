FROM rocker/r-ver:4.3.2

LABEL org.opencontainers.image.title="marlin" \
      org.opencontainers.image.description="MARLIN runtime environment (MARLIN + modkit + R keras3 + TensorFlow CPU)" \
      org.opencontainers.image.source="https://github.com/hovestadt/MARLIN"

ENV DEBIAN_FRONTEND=noninteractive \
    LANG=C.UTF-8 \
    TZ=Asia/Tokyo

# 1. OS パッケージ
RUN apt-get update && apt-get install -y --no-install-recommends \
        ca-certificates \
        curl \
        wget \
        git \
        gzip \
        tar \
        tzdata \
        procps \
        samtools \
        libicu-dev \
        libxml2-dev \
        libssl-dev \
        libcurl4-openssl-dev \
        zlib1g-dev \
        libbz2-dev \
        liblzma-dev \
        libdeflate-dev \
        libpng-dev \
        libjpeg-dev \
        libtiff5-dev \
        libcairo2-dev \
        libfreetype6-dev \
        libfontconfig1-dev \
        libharfbuzz-dev \
        libfribidi-dev \
        python3 \
        python3-dev \
        python3-pip \
        python3-venv && \
    rm -rf /var/lib/apt/lists/*

# modkit
RUN wget -q https://github.com/nanoporetech/modkit/releases/download/v0.5.0/modkit_v0.5.0_u16_x86_64.tar.gz && \
    tar xvf modkit_v0.5.0_u16_x86_64.tar.gz && \
    install -m 0755 dist_modkit_v0.5.0_5120ef7/modkit /usr/local/bin/

# MARLIN
ARG MARLIN_REPO=https://github.com/hovestadt/MARLIN.git
RUN wget -q https://github.com/hovestadt/MARLIN/archive/refs/tags/v1.0.tar.gz && \
    tar xvf v1.0.tar.gz 

# TensorFlow (CPU)
RUN pip install --no-cache-dir --upgrade pip setuptools wheel && \
    pip install --no-cache-dir \
        "numpy==1.26.4" \
        "scipy==1.11.4" \
        "h5py>=3.10" \
        "tensorflow-cpu==2.16.2" \
        "keras==3.3.3" \
        "pandas" "Pillow" "pydot" "ipython"
        
# R パッケージ
RUN Rscript -e 'install.packages(c("data.table", "openxlsx", "reticulate", "keras3", "doParallel", "foreach"), repos="https://cloud.r-project.org", Ncpus=parallel::detectCores())'

# reticulate に uv を使わせない
ENV RETICULATE_PYTHON=/opt/venv/bin/python \
    RETICULATE_USE_MANAGED_VENV=no \
    KERAS_BACKEND=tensorflow \
    KERAS_HOME=/opt/keras \
    TF_CPP_MIN_LOG_LEVEL=2 \
    TF_ENABLE_ONEDNN_OPTS=0
