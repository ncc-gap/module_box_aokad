FROM friend1ws/nanomonsv:v0.8.0

LABEL maintainer="Ai Okada <aokada@ncc.go.jp>"

RUN apt-get update && apt-get install -y --no-install-recommends \
        ca-certificates curl wget bzip2 \
        build-essential \
        zlib1g-dev libbz2-dev liblzma-dev libcurl4-openssl-dev libncurses-dev \
    && rm -rf /var/lib/apt/lists/*


RUN curl -fsSL -O https://github.com/samtools/samtools/releases/download/1.21/samtools-1.21.tar.bz2 \
    && tar xjf samtools-1.21.tar.bz2 \
    && cd samtools-1.21 \
    && ./configure --prefix=/usr/local --with-htslib=/usr/local \
    && make && make install

RUN curl -fsSL -O https://github.com/shenwei356/seqkit/releases/download/v2.14.0/seqkit_linux_amd64.tar.gz \
    && tar xzf seqkit_linux_amd64.tar.gz \
    && install -m 0755 seqkit /usr/local/bin/seqkit
